## implement full-text search
!think Search stays out of the store: a function over store.all() keeps it independent of the tags work.
!tool read_file {"path": "src/store.py"}
!then
!tool write_file {"path": "src/search.py", "content": "\"\"\"Full-text search over a NoteStore.\"\"\"\n\n\ndef search(store, query):\n    \"\"\"Notes whose title or body contains `query`, ignoring case.\"\"\"\n    needle = query.lower()\n    return [\n        note\n        for note in store.all()\n        if needle in note[\"title\"].lower() or needle in note[\"body\"].lower()\n    ]\n"}
!tool write_file {"path": "tests/test_search.py", "content": "import pathlib\nimport sys\nimport unittest\n\nsys.path.insert(0, str(pathlib.Path(__file__).parent.parent / \"src\"))\nfrom search import search\nfrom store import NoteStore\n\n\nclass SearchTest(unittest.TestCase):\n    def test_matches_title_and_body_ignoring_case(self):\n        store = NoteStore()\n        milk = store.add(\"Buy MILK\")\n        note = store.add(\"Groceries\", body=\"milk, eggs\")\n        store.add(\"Unrelated\")\n        self.assertEqual(search(store, \"milk\"), [milk, note])\n\n\nif __name__ == \"__main__\":\n    unittest.main()\n"}
!then
!tool bash {"command": "python3 -m unittest discover -s tests"}
!then
!say Search is implemented and the suite passes (4 tests).
!say
